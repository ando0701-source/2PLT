# Fix: add Line / Column / LineOffset while keeping source-wide Offset.
# Previous fix retained: prevent identifiers/keywords from being misclassified as DecimalFloatingPoint.

function Get-JavaTokens {
    <#
    .SYNOPSIS
        Java ソースコードをトークン分割します（Java 21 の主要字句に対応）。

    .PARAMETER Path
        Java ソースファイルのパス。

    .PARAMETER Code
        直接 Java コード文字列を渡す場合。

    .PARAMETER IncludeComments
        コメントトークンも結果に含める場合に指定します。

    .PARAMETER Detailed
        トークン文字列ではなく、Value / Type / SubType / Line / Column /
        LineOffset / Offset / Length を持つオブジェクトとして返します。

        Line       : 1 始まりの行番号
        Column     : 1 始まりの列番号
        LineOffset : 0 始まりの行頭からの文字オフセット
        Offset     : 0 始まりのソース全体先頭からの文字オフセット

    .OUTPUTS
        Detailed 未指定: [string[]]
        Detailed 指定  : [PSCustomObject[]]

    .NOTES
        - Contextual Keyword は構文上の文脈がないと確定できないため、
          ContextualKeywordCandidate として出力します。
        - Unicode escape の事前変換や Java 構文解析までは行いません。
        - type context における > / >> / >>> の構文依存分割は行いません。
        - Java 21 preview の String Template は簡易認識です。
    #>

    param(
        [Parameter(Mandatory = $false, Position = 0)]
        [string] $Path,

        [Parameter(Mandatory = $false)]
        [string] $Code,

        [switch] $IncludeComments,

        [switch] $Detailed
    )

    if (-not $PSBoundParameters.ContainsKey('Code')) {
        if ([string]::IsNullOrWhiteSpace($Path)) {
            throw 'Path または Code のいずれかを指定してください。'
        }

        if (-not (Test-Path -LiteralPath $Path -PathType Leaf)) {
            throw "ファイルが存在しません: $Path"
        }

        $Code = Get-Content -LiteralPath $Path -Raw -Encoding UTF8
    }

    # Java 21 Reserved Keywords
    $reservedKeywords = @(
        'abstract', 'assert', 'boolean', 'break', 'byte',
        'case', 'catch', 'char', 'class', 'const',
        'continue', 'default', 'do', 'double', 'else',
        'enum', 'extends', 'final', 'finally', 'float',
        'for', 'goto', 'if', 'implements', 'import',
        'instanceof', 'int', 'interface', 'long', 'native',
        'new', 'package', 'private', 'protected', 'public',
        'return', 'short', 'static', 'strictfp', 'super',
        'switch', 'synchronized', 'this', 'throw', 'throws',
        'transient', 'try', 'void', 'volatile', 'while',
        '_'
    )

    # Java 21 Contextual Keywords
    # non-sealed は構文文脈が必要なため、単純な正規表現では常時1トークン扱いしない。
    $contextualKeywords = @(
        'exports', 'module', 'open', 'opens', 'permits',
        'provides', 'record', 'requires', 'sealed', 'to',
        'transitive', 'uses', 'var', 'when', 'with', 'yield'
    )

    # Java 21 の主要字句を longest-match 優先で抽出する。
    # (?x) により空白と # コメントを正規表現中で無視する。
    $javaTokenPattern = @'
(?x)
(
      # --- コメント ---
      //[^\r\n]*
    | /\*[\s\S]*?\*/

      # --- Java 21 preview String Template（簡易認識） ---
    | [\p{L}\p{Nl}\p{Sc}\p{Pc}][\p{L}\p{Nl}\p{Sc}\p{Pc}\p{Nd}\p{Mc}\p{Mn}\p{Cf}]*\."""[\s\S]*?"""
    | [\p{L}\p{Nl}\p{Sc}\p{Pc}][\p{L}\p{Nl}\p{Sc}\p{Pc}\p{Nd}\p{Mc}\p{Mn}\p{Cf}]*\."(?:[^"\\\r\n]|\\.)*"

      # --- テキストブロック / 文字列 / 文字 ---
    | """[\s\S]*?"""
    | "(?:[^"\\\r\n]|\\.)*"
    | '(?:[^'\\\r\n]|\\.)'

      # --- 16進浮動小数点リテラル ---
    | 0[xX](?:
          [0-9a-fA-F](?:[0-9a-fA-F_]*[0-9a-fA-F])?\.?
        | (?:[0-9a-fA-F](?:[0-9a-fA-F_]*[0-9a-fA-F])?)?\.[0-9a-fA-F](?:[0-9a-fA-F_]*[0-9a-fA-F])?
      )[pP][+-]?[0-9](?:[0-9_]*[0-9])?[fFdD]?

      # --- 10進浮動小数点リテラル ---
    | (?:
          [0-9](?:[0-9_]*[0-9])?\.(?:[0-9](?:[0-9_]*[0-9])?)?(?:[eE][+-]?[0-9](?:[0-9_]*[0-9])?)?[fFdD]?
        | \.[0-9](?:[0-9_]*[0-9])?(?:[eE][+-]?[0-9](?:[0-9_]*[0-9])?)?[fFdD]?
        | [0-9](?:[0-9_]*[0-9])?[eE][+-]?[0-9](?:[0-9_]*[0-9])?[fFdD]?
        | [0-9](?:[0-9_]*[0-9])?[fFdD]
      )

      # --- 整数リテラル ---
    | 0[xX][0-9a-fA-F](?:[0-9a-fA-F_]*[0-9a-fA-F])?[lL]?
    | 0[bB][01](?:[01_]*[01])?[lL]?
    | 0(?:_*[0-7])(?:[0-7_]*[0-7])?[lL]?
    | (?:0|[1-9](?:[0-9_]*[0-9])?)[lL]?

      # --- 識別子（Java identifier の主要 Unicode category を対象） ---
    | [\p{L}\p{Nl}\p{Sc}\p{Pc}][\p{L}\p{Nl}\p{Sc}\p{Pc}\p{Nd}\p{Mc}\p{Mn}\p{Cf}]*

      # --- 複合演算子（longest-match 優先） ---
    | >>>= | >>= | <<= | >>> | >> | <<
    | == | != | <= | >= | && | \|\| | \+\+ | --
    | \+= | -= | \*= | /= | &= | \|= | \^= | %=
    | ->

      # --- 複合区切り記号 ---
    | \.\.\. | ::

      # --- 単一演算子 ---
    | [=>>>=|>>=|<<=|>>>|>>|<<|==|!=|<=|>=|&&|\|\||\+\+|--|\+=|-=|\*=|/=|&=|\|=|\^=|%=|->|[=>
